\par Our residuals are this:
\begin{equation}
S = \sum_{p\neq q} \sum_{mn} d_{pq}^{mn} - g_p^{ma} M^{ab} \overline{g_q^{bn}}.
\end{equation}
$p$ and $q$ are baselines, $m$ and $n$ are correlations ($\in {r,l}$). Repeated indices are summed. 
What we minimize is the norm of $S$, that is, $S\overline{S}$, or $|S|^2$.
To this end, we compute gradient of the loss, that is, $\frac{\partial |S|^2}{\partial g_p^{mn}}$ for every $m,n$. 

\begin{equation}
\begin{split}

d_{pq}^{rr} &\left(- g_q^{lr} \overline{m^{rl}} - g_q^{rr} \overline{m^{rr}}\right) + \\
d_{pq}^{rl} &\left(- g_q^{ll} \overline{m^{rl}} - g_q^{rl} \overline{m^{rr}}\right) + \\
g_p^{rr} &\left(g_q^{ll} m^{rl} \overline{g_q^{ll}} \overline{m^{rl}} + g_q^{ll} m^{rr} \overline{g_q^{rl}} \overline{m^{rl}} + g_q^{lr} m^{rl} \overline{g_q^{lr}} \overline{m^{rl}} + g_q^{lr} m^{rr} \overline{g_q^{rr}} \overline{m^{rl}} + g_q^{rl} m^{rl} \overline{g_q^{ll}} \overline{m^{rr}} + g_q^{rl} m^{rr} \overline{g_q^{rl}} \overline{m^{rr}} + g_q^{rr} m^{rl} \overline{g_q^{lr}} \overline{m^{rr}} + g_q^{rr} m^{rr} \overline{g_q^{rr}} \overline{m^{rr}}\right) + \\

\end{split}
\end{equation}
