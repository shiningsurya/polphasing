\par Our residuals are this:
\begin{equation}
S = \sum_{p\neq q} \sum_{mn} d_{pq}^{mn} - g_p^{mm} M^{mn} \overline{g_q^{nn}}.
\end{equation}
In this equation, we assume the gains are parallel.
$p$ and $q$ are baselines, $m$ and $n$ are correlations ($\in {r,l}$). Repeated indices are summed. 
What we minimize is the norm of $S$, that is, $S\overline{S}$, or $|S|^2$.
To this end, we compute gradient of the loss, that is, $\frac{\partial |S|^2}{\partial g_p^{mn}}$ for every $m,n$. 

\begin{equation}
\begin{split}
\frac{\partial |S|^2}{\partial g_p^{rr}} &= \sum_q \\ 
- &d_{pq}^{rl} g_q^{ll} \overline{m^{rl}}+\\
- &d_{pq}^{rr} g_q^{rr} \overline{m^{rr}}+\\
	&g_p^{rr} \left(g_q^{ll} m^{rl} \overline{g_q^{ll}} \overline{m^{rl}} + g_q^{rr} m^{rr} \overline{g_q^{rr}} \overline{m^{rr}}\right)
\end{split}
\end{equation}

\begin{equation}
\begin{split}
\frac{\partial |S|^2}{\partial g_p^{ll}} &= \sum_q \\ 
- &d_{pq}^{ll} g_q^{ll} \overline{m^{ll}}+\\
- &d_{pq}^{lr} g_q^{rr} \overline{m^{lr}}+\\
	&g_p^{ll} \left(g_q^{ll} m^{ll} \overline{g_q^{ll}} \overline{m^{ll}} + g_q^{rr} m^{lr} \overline{g_q^{rr}} \overline{m^{lr}}\right)
\end{split}
\end{equation}
